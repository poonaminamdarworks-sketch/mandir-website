<script>
function toggleLang(){
document.querySelectorAll('[data-en]').forEach(el=>{
el.innerText = el.innerText === el.getAttribute('data-en')
? el.getAttribute('data-mr')
: el.getAttribute('data-en');
});
}

function filter(type){
let items=document.querySelectorAll('#galleryGrid .card');
items.forEach(i=>{
if(type==='all'){i.style.display='block'}
else{
i.style.display=i.classList.contains(type)?'block':'none'
}
})
}
</script>
